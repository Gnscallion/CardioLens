# --- 运行时必需 ---

# Web 框架：应用工厂 + 蓝图 + 请求上下文
flask>=3.0

# 生产 WSGI 服务器。选它而不是 gunicorn 有两个原因：
#   1) gunicorn 依赖 fork，在 Windows 上根本跑不起来；waitress 纯 Python、跨平台
#   2) 本应用的限流计数与并发信号量都是进程内状态，必须单进程多线程运行，
#      而 waitress 默认就是单进程多线程，正好契合
waitress>=3.0

# 深度学习（推理必需；训练同样使用）。
# 本项目在纯 CPU 上即可训练与推理，务必使用 CPU 版，不要升级到 CUDA 版：
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
torch>=2.0

# 数值计算（推理必需）
numpy>=1.24


# ---------------------------- 以下仅在“从零重建数据集 / 重新训练”时需要 ----------------------------

matplotlib>=3.7        # 生成 models/confusion_matrix.png 与 training_curve.png（train 绘图用）
